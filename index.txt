  $reversed='';
  $i=0;
  while(isset($str[$i])){
    $reversed=$str[$i].$reversed;
    $i++;
  }
  return $reversed;

<!-----------13. palindrome string sentance ------------->

<?php
function rev($str){
  $words = explode(' ', $str);  
  $reversed = '';
  $i = count($words) - 1;      

  while ($i >= 0) {
    $reversed .= $words[$i];   
    if ($i > 0) $reversed .= ' '; 
    $i--;
  }

  return $reversed;
}

$input = "hello well what jhjh";
$reversed = rev($input);
echo $reversed;
?>
